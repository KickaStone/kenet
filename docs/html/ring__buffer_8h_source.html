12// SPSC ring
13template <typename T, size_t CapacityPow2>
14class SpscRing {
15 static_assert((CapacityPow2 & (CapacityPow2 - 1)) == 0, "Capacity must be power of 2");
16 static_assert(std::is_trivially_copyable<T> :: value, "T must be trivially copyable");
17 static constexpr size_t MASK = CapacityPow2 - 1;
18public:
19 bool try_push(const T& v) {
20 auto w = write_.load(std::memory_order_relaxed);
21 auto n = w + 1;
22 if (n==read_.load(std::memory_order_acquire)) return false; // full
23 buf_[w & MASK] = v;
24 write_.store(n, std::memory_order_release);
25 return true;
26 }
27
28 bool try_pop(T& out) {
29 auto r = read_.load(std::memory_order_relaxed);
30 if (r == write_.load(std::memory_order_acquire)) return false; // empty
31 out = buf_[r & MASK];
32 read_.store(r + 1, std::memory_order_release);
33 return true;
34 }
35
36 bool empty() const {
37 return read_.load(std::memory_order_acquire) == write_.load(std::memory_order_acquire);
38 }
39private:
40 alignas(64) std::atomic<size_t> write_{0};
41 alignas(64) std::atomic<size_t> read_{0 };
42 alignas(64) std::array<T, CapacityPow2> buf_{};
43};
