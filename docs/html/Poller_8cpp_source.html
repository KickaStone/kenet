40int Poller::poll(int timeoutMs, std::vector<Channel*>& active) {
41 int nfds = epoll_wait(epfd_, events_, MAX_EVENTS, timeoutMs);
42 if (nfds < 0) {

46 for (int i = 0; i < nfds; ++i) {
47 Channel* ch = static_cast<Channel*>(events_[i].data.ptr);
