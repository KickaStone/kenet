59int Poller::poll(int timeoutMs, std::vector<Channel*>& active) {
60 if (timeoutMs <= 0) {
61 timeoutMs = -1; // 如果timeoutMs小于等于0，则设置为-1，表示无限等待。
62 }
63 int nfds = epoll_wait(epfd_, events_, MAX_EVENTS, timeoutMs);
64 if (nfds < 0) {
65 LOG_ERROR("epoll_wait failed");
66 return -1;
67 }
68 for (int i = 0; i < nfds; ++i) {
69 Channel* ch = static_cast<Channel*>(events_[i].data.ptr);
70 ch->setRevents(events_[i].events);
71 active.push_back(ch);
72 }
73 return nfds;
74}
