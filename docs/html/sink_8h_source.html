39// ---- FileSink: 追加写 & 可控 fsync ----
40class FileSink : public ISink {
41public:
42 explicit FileSink(std::string path, bool fsync_on_flush, size_t rotate_bytes)
43 : path_(std::move(path)), fsync_on_flush_(fsync_on_flush), rotate_bytes_(rotate_bytes) {
44 open_file();
45 }
46
47 ~FileSink() override { if (fd_ >= 0) ::close(fd_); }
48
49 void write(LineView lv) override {
50 if (lv.level < min_level) return;
51 buf_.append(lv.data, lv.size);
52 written_ += lv.size;
53 if (written_ >= rotate_bytes_) rotate();
54 }
55
56 void flush(bool final) override {
57 if (buf_.empty()) {
58 if (final && fsync_on_flush_ && fd_ >= 0) ::fsync(fd_);
59 return;
60 }
61 if (const ssize_t bytes_written = ::write(fd_, buf_.data(), buf_.size()); bytes_written == -1) {
62 perror("write failed.");
63 }else if (bytes_written < static_cast<ssize_t>(buf_.size())) {
64 std::cerr << "Warning: Only " << bytes_written << " out of " << " bytes written." << std::endl;
65 // do nothing, only record
66 }
67 if (fsync_on_flush_ || final) ::fsync(fd_);
68 buf_.clear();
69 }
70
71private:
72 void open_file() {
73 if (fd_ >= 0) ::close(fd_);
74 fd_ = ::open(path_.c_str(), O_CREAT | O_APPEND | O_WRONLY | O_CLOEXEC, 0644);
75 written_ = 0;
76 }
77
78 void rotate() {
79 // 简单基于时间戳的轮转
80 char tbuf[32];
81 time_t sec = time(nullptr);
82 tm tmv;
83 localtime_r(&sec, &tmv);
84 snprintf(tbuf, sizeof(tbuf), "%04d%02d%02d-%02d%02d%02d",
85 tmv.tm_year + 1900, tmv.tm_mon + 1, tmv.tm_mday,
86 tmv.tm_hour, tmv.tm_min, tmv.tm_sec);
87 char newname[1024];
88 snprintf(newname, sizeof(newname), "%s.%s.%d", path_.c_str(), tbuf, (int) getpid());
89 ::close(fd_);
90 ::rename(path_.c_str(), newname);
91 open_file();
92 }
93
94 std::string path_;
95 bool fsync_on_flush_{false};
96 size_t rotate_bytes_{128ull << 20};
97
98 int fd_{-1};
99 std::string buf_;
100 size_t written_{0};
101};

103// ---- ConsoleSink: 输出到 stderr，支持 ANSI 颜色 ----
104class ConsoleSink : public ISink {
105public:
106 explicit ConsoleSink(bool enable_color = true): color_(enable_color) {
107 }
108
109 void write(LineView lv) override {
110 if (lv.level < min_level) return;
111 if (!color_) {
112 buf_.append(lv.data, lv.size);
113 return;
114 }
115 // 简单着色：Warn/Yellow, Error/Red, Fatal/Bold Red
116 const char *pre = "", *suf = "\033[0m";
117 switch (lv.level) {
118 case Level::Warn: pre = "\033[33m";
119 break;
120 case Level::Error: pre = "\033[31m";
121 break;
122 case Level::Fatal: pre = "\033[1;31m";
123 break;
124 default: pre = "";
125 suf = "";
126 break;
127 }
128 if (*pre) buf_.append(pre);
129 buf_.append(lv.data, lv.size);
130 if (*pre) buf_.append(suf);
131 }
132
133 void flush(bool) override {
134 if (buf_.empty()) return;
135 if (const ssize_t bytes_written = ::write(STDOUT_FILENO, buf_.data(), buf_.size()); bytes_written == -1) {
136 perror("write failed.");
137 }else if (bytes_written < static_cast<ssize_t>(buf_.size())) {
138 std::cerr << "Warning: Only " << bytes_written << " out of " << " bytes written." << std::endl;
139 }
140 buf_.clear();
141 }

143private:
144 bool color_{true};

146};
