123 // 可在任意线程调用
124 void logf(Level lv, const char* fmt, ...) {
125 auto* prod = get_or_create_producer();
126 Entry e{};
127 e.level = static_cast<Level>(lv);
128 e.tid = thread_id_hash();
129 e.ts_ns = now_ns_coarse();
130
131 char buf[MaxLine];
132 va_list ap; va_start(ap, fmt);
133 int n = vsnprintf(buf, sizeof(buf), fmt, ap);
134 va_end(ap);
135 if (n < 0) return;
136
137 if (static_cast<size_t>(n) >= sizeof(buf)) {
138 // 截断并标注
139 constexpr const char* tail = "...(trunc)";
140 size_t keep = sizeof(buf) - 1;
141 size_t tl = strlen(tail);
142 if (keep > tl) {
143 memcpy(buf + keep - tl, tail, tl);
144 }
145 n = static_cast<int>(keep);
146 }
147 e.len = static_cast<uint16_t>(n);
148 memcpy(e.msg, buf, e.len);
149
150 if (!prod->ring->try_push(e)) {
151 prod->dropped.fetch_add(1, std::memory_order_relaxed);
152 } else {
153 // 轻量唤醒：避免纯自旋
154 if (++wake_hint_ % 1024 == 0) cv_.notify_one();
155 }
156 }

179 static uint64_t now_ns_coarse() {
180#ifdef CLOCK_REALTIME_COARSE
181 timespec ts; clock_gettime(CLOCK_REALTIME_COARSE, &ts);
182#else
183 timespec ts; clock_gettime(CLOCK_REALTIME, &ts);

185 return uint64_t(ts.tv_sec) * 1000000000ull + uint64_t(ts.tv_nsec);

192 Producer* get_or_create_producer() {
193 thread_local Producer* tls = nullptr;
194 if (tls) return tls;
195 auto p = std::make_unique<Producer>();
196 tls = p.get();

205 // 后台线程
206 void run() {
207 uint64_t last_flush_ns = now_ns_coarse();
208 size_t bytes_since_flush = 0;
209
210 while (true) {
211 if (!running_) {
212 drain_all(bytes_since_flush);
213 // final flush
214 file_sink_->flush(true);
215 console_sink_->flush(true);
216 break;
217 }
218
219 bool did = drain_all(bytes_since_flush);
220
221 uint64_t now = now_ns_coarse();
222 bool time_to_flush =
223 (now - last_flush_ns) >= (uint64_t(cfg_.flush_interval.count()) * 1000000ull);
224
225 if (bytes_since_flush >= cfg_.flush_bytes || time_to_flush) {
226 file_sink_->flush(false);
227 console_sink_->flush(false);
228 bytes_since_flush = 0;
229 last_flush_ns = now;
230 }
231
232 if (!did) {
233 std::unique_lock<std::mutex> lk(mu_);
234 cv_.wait_for(lk, std::chrono::microseconds(200));
235 }
236 }
237 }

239 bool drain_all(size_t& bytes_since_flush) {
240 bool did = false;
241 std::vector<Producer*> local;
242 {
243 std::lock_guard<std::mutex> lk(mu_);
244 local = producers_;
245 }
246 for (auto* p : local) {
247 Entry e;
248 int round = 0;
249 while (p->ring->try_pop(e)) {
250 did = true;
251 // 一次格式化，多路写
252 std::string line = format_line(e);
253 LineView lv{ line.data(), line.size(), static_cast<Level>(e.level) };
254 file_sink_->write(lv);
255 console_sink_->write(lv);
256 bytes_since_flush += lv.size;
257 if (++round >= 1024) break;
258 }
259 }
260 return did;
261 }
